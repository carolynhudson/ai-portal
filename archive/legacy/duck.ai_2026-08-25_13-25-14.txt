log_success() { echo -e "${GREEN}[✓]${NC} $1"; }
log_warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }

ensure_dir() {
    if [ ! -d "$1" ]; then

check_rocm() {

        if [ "$gpu_count" -gt 0 ]; then
            log_success "ROCM detected with GPU support"
            rocm-smi --showproductname
            return 0
        fi
    fi
    log_warn "ROCM not properly detected, will attempt CPU-only or HIP compilation"
    return 1
}

create_systemd_service() {
    cat > "$SYSTEMD_FILE" << EOF
[Unit]
Description=llama.cpp Server with ROCM Acceleration
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=$SERVICE_USER
WorkingDirectory=$LLAMA_INSTALL_DIR

# Load configuration
EnvironmentFile=$CONFIG_DIR/llamacpp-server.conf

# Main command - load default model
ExecStart=$LLAMA_BIN_DIR/llamacpp-server \\
  --model "\${DEFAULT_MODEL}" \\
  --port \${PORT} \\
  --host \${HOST} \\
  --ctx-size \${CONTEXT_SIZE} \\
  --batch-size \${BATCH_SIZE} \\
  --threads \${THREADS} \\
  --n-gpu-layers \${GPU_LAYERS} \\
  --verbose

# Restart policy
Restart=on-failure
RestartSec=10
StandardOutput=journal
StandardError=journal
SyslogIdentifier=llamacpp

# Resource limits
LimitNOFILE=65536
LimitNPROC=65536

[Install]
WantedBy=default.target
EOF

    chmod 644 "$SYSTEMD_FILE"
    
    # Reload systemd
    systemctl --user daemon-reload
    systemctl --user enable "$SERVICE_NAME"
    
    log_success "Created and enabled systemd service: $SERVICE_NAME"
}

    systemctl --user enable "$SERVICE_NAME" 2>/dev/null || true
    log_success "Systemd service: $SERVICE_NAME"
}

create_helpers() {
    cat > "$LLAMA_BIN_DIR/llamacpp-start-model" << 'HELPEREOF'
#!/bin/bash
if [ -z "$1" ]; then
    echo "Usage: llamacpp-start-model <model-path>"
    exit 1
fi
if [ ! -f "$1" ]; then
    echo "Model not found: $1"
    exit 1
fi
source ~/.config/llama-cpp/llamacpp-server.conf
$HOME/.local/bin/llamacpp-server --model "$1" --port $PORT --host $HOST --ctx-size $CONTEXT_SIZE --batch-size $BATCH_SIZE --threads $THREADS --n-gpu-layers $GPU_LAYERS --verbose
HELPEREOF
    chmod +x "$LLAMA_BIN_DIR/llamacpp-start-model"
    log_success "Helper: llamacpp-start-model"
    
    cat > "$LLAMA_BIN_DIR/llamacpp-update" << 'UPDATEEOF'
#!/bin/bash
set -e
INSTALL_DIR="${HOME}/.local/share/llama-cpp"
echo "Stopping service..."
systemctl --user stop llamacpp-server 2>/dev/null || true
cd "$INSTALL_DIR"
echo "Pulling updates..."
git pull origin master || exit 1
echo "Rebuilding..."
rm -rf build && mkdir -p build && cd build
cmake .. -DCMAKE_BUILD_TYPE=Release -DLLAMA_ROCM=ON -DAMDGPU_TARGETS=gfx942 || exit 1
cmake --build . --config Release -j "$(nproc)" || exit 1
echo "Installing..."
BIN_DIR="${HOME}/.local/bin"
cd "$INSTALL_DIR"
[ -f "build/bin/server" ] && cp "build/bin/server" "$BIN_DIR/llamacpp-server"
[ -f "build/bin/main" ] && cp "build/bin/main" "$BIN_DIR/llamacpp-cli"
[ -f "build/bin/quantize" ] && cp "build/bin/quantize" "$BIN_DIR/llamacpp-quantize"
systemctl --user start llamacpp-server
echo "✓ Updated successfully!"
UPDATEEOF
    chmod +x "$LLAMA_BIN_DIR/llamacpp-update"
    log_success "Helper: llamacpp-update"
}
